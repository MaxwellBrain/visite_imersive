% ============================== SOMMAIRE ET TABLE DES MATIÈRES : LES POINTS ==
% La classe « report » prive les lignes de chapitre de points de conduite : le
% titre reste à gauche, le numéro de page à droite, et rien entre les deux. Le
% sommaire devenait illisible dès que les titres étaient de longueurs
% inégales. On redéfinit donc \l@chapter pour :
%   · tirer des points du titre jusqu'au numéro de page ;
%   · préfixer « CHAPITRE n : » sur les chapitres numérotés, comme dans le
%     corps du document — les liminaires, non numérotés, gardent leur titre seul.
\makeatletter
\renewcommand*\l@chapter[2]{%
  \ifnum \c@tocdepth >\m@ne
    \addpenalty{-\@highpenalty}%
    \vskip 1.0em \@plus\p@
    \setlength\@tempdima{1.6em}%
    \begingroup
      \parindent \z@ \rightskip \@pnumwidth
      \parfillskip -\@pnumwidth
      \leavevmode \bfseries \color{keyceblue}%
      \advance\leftskip\@tempdima
      \hskip -\leftskip
      % « \numberline{3} » devient « CHAPITRE 3 : »
      \renewcommand*\numberline[1]{\hb@xt@\@tempdima{\hss}CHAPITRE~##1~: }%
      % Un blanc rigide avant les points : sans lui, un titre long colle au
      % numero de page et la ligne se retrouve sans points de conduite.
      #1\nobreak\hskip 1.2em\nobreak
      \leaders\hbox{$\m@th\mkern\@dotsep mu\hbox{.}\mkern\@dotsep mu$}\hfill
      \nobreak\hb@xt@\@pnumwidth{\hss #2}\par
      \penalty\@highpenalty
    \endgroup
  \fi}
\makeatother

% ================================================== TITRES (CHAPITRES, ETC.)
% Les grands titres sont CENTRÉS dans un rectangle à coins arrondis, comme dans
% le rapport BTS (forme « flowChartAlternateProcess » de Word, contour accent1,
% fond en teinte claire du même bleu).
\newcommand{\cadretitre}[1]{%
  \begin{center}
  \begin{tcolorbox}[
      enhanced, nobeforeafter,
      colback=keycelight, colframe=keyceaccent,
      boxrule=1.1pt, arc=7pt,
      width=0.90\textwidth,
      % Hauteur PLANCHER et non hauteur fixe. Le plancher de 1,15 cm garantit que
      % tous les cadres d'une seule ligne (RÉSUMÉ, DÉDICACE, titres courts) aient
      % exactement la même taille ; au-delà, le cadre GRANDIT avec le titre au lieu
      % de le laisser déborder, ce qui se produisait sur les chapitres 3 et 4.
      height from=1.15cm to 4.2cm,
      left=8pt, right=8pt, top=4pt, bottom=4pt,
      halign=center, valign=center]
    {\normalfont\bfseries\fontsize{14}{17}\selectfont\color{keyceblue}%
     % Interligne resserré et coupure de mots interdite : un titre long tient
     % alors sur deux lignes propres, sans césure du type « FÉLI-CIEN ».
     \setstretch{1.02}%
     \hyphenpenalty=10000\exhyphenpenalty=10000
     \MakeUppercase{#1}\par}
  \end{tcolorbox}
  \end{center}}
